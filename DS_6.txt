// Experiment No. 6
#include <stdio.h>
#include <string.h>
#define SIZE 10

struct Entry
{
    int key;
    char name[100];
};

struct Entry table[SIZE];

// function prototypes
void initialize();
void insert();
void display();
int find(int k);
void Delete(int k);

int main()
{
    int k;
    int ch, index;
    char ans;

    initialize();

    do {
        printf("\n\t***** Hash Table Menu *****");
        printf("\n\t1. Insert\n\t2. Display\n\t3. Find\n\t4. Delete\n\t5. Exit");
        printf("\n\t..... Enter Your Choice: ");
        scanf("%d", &ch);

        switch (ch)
        {
            case 1:
                insert();
                break;

            case 2:
                display();
                break;

            case 3:
                printf("\n\tEnter a Key to Search: ");
                scanf("%d", &k);
                index = find(k);
                if (index == -1)
                {
                    printf("\n\tKey Not Found!");
                }
                break;

            case 4:
                printf("\n\tEnter a Key to Delete: ");
                scanf("%d", &k);
                Delete(k);
                break;

            case 5:
                printf("\n\tExiting...");
                return 0;

            default:
                printf("\n\tInvalid Choice!");
        }

        printf("\n\t..... Do You Want to Continue in Main Menu (y/n): ");
        scanf(" %c", &ans);

    } while (ans == 'y' || ans == 'Y');

    return 0;
}

// initialize table with -1
void initialize()
{
    for (int i = 0; i < SIZE; i++)
    {
        table[i].key = -1;  // empty
        strcpy(table[i].name, "NULL");
    }
}

// insert key and value
void insert()
{
    char ans;
    int k;
    char n[100];

    do {
        printf("\n\tEnter a Key: ");
        scanf("%d", &k);
        printf("\n\tEnter a Value: ");
        scanf("%99s", n);

        int hi = k % SIZE;
        int flag = 0;

        for (int i = 0; i < SIZE; i++)
        {
            int index = (hi + i) % SIZE;
            if (table[index].key == -1)
            {
                table[index].key = k;
                strcpy(table[index].name, n);
                printf("\n\tInserted at index %d", index);
                flag = 1;
                break;
            }
        }

        if (flag == 0)
        {
            printf("\n\tHash Table is FULL!");
        }

        printf("\n\t..... Do You Want to Insert More Key (y/n): ");
        scanf(" %c", &ans);

    } while (ans == 'y' || ans == 'Y');
}

// display hash table
void display()
{
    printf("\n\tIndex\tKey\t\tName");
    for (int i = 0; i < SIZE; i++)
    {
        printf("\n\th[%d]\t", i);
        printf("%d\t\t%s", table[i].key, table[i].name);
    }
    printf("\n");
}

// search for a key
int find(int k)
{
    int hi = k % SIZE;
    for (int i = 0; i < SIZE; i++)
    {
        int index = (hi + i) % SIZE;
        if (table[index].key == k)
        {
            printf("\n\t%d is Found at index %d with Name %s", k, index, table[index].name);
            return index;
        }
        if (table[index].key == -1)
            return -1;
    }
    return -1;
}

void Delete(int k)
{
    int index = find(k);
    if (index == -1)
    {
        printf("\n\tKey Not Found!");
    }
    else
    {
        table[index].key = -1;  // mark as deleted
        strcpy(table[index].name, "NULL");
        printf("\n\tKey %d Deleted!", k);
    }
}
